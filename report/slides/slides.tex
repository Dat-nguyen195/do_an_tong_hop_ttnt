\begin{frame}{1. Ambiguous specifications}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Specification: ``Remove duplicate elements.''}
      \par\medskip
      \begin{itemize}
        \item Stable first occurrences: \texttt{[4, 1, 2]}.
        \item Sorted unique values: \texttt{[1, 2, 4]}.
      \end{itemize}
      \par\medskip
      Both interpretations fit the description for input \texttt{[4, 1, 4, 2, 1]}.
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{SEMSYNTH interaction}
        Generate a candidate pool, select a distinguishing question, and revise the pool after a Yes/No answer.
      \end{block}
      \par\medskip
      The original design aims to reduce the effort of expressing intent. This project evaluates candidate selection; human effort has not been measured.
    \end{column}

\begin{frame}{2. Original SEMSYNTH architecture}
  \begin{enumerate}
    \item \textbf{Specification parsing:} goals, constraints, and interface.
    \item \textbf{Feature extraction:} 47 code, execution, and question features.
    \item \textbf{AGA initialization:} generate strategies, then translate to code.
    \item \textbf{Question selection:} semantic information gain and diversity.
    \item \textbf{Oracle interaction:} atomic Yes/No clarification.
    \item \textbf{Belief revision:} Bayesian updates and consistency checks.
    \item \textbf{Termination / regeneration:} select a candidate or rebuild the pool.
  \end{enumerate}
  \par\medskip
  \textbf{Course scope:} a bounded execution-based subset of this architecture.
  \source{Supplied SEMSYNTH manuscript; course report, Method}
\end{frame}

\begin{frame}{3. Original feature model and question design}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{47-dimensional representation}
      \begin{itemize}
        \item 18 AST structural features.
        \item 18 execution and semantic features.
        \item 11 question-text features.
      \end{itemize}
      \par\medskip
      XGBoost predicts satisfaction:
      \[
        M(h,q)=P(h\models q).
      \]
      \textbf{Not trained or integrated} in this prototype.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Semantic question design}
      \begin{itemize}
        \item 85 templates in nine clusters.
        \item Atomic decomposition of compound queries.
        \item Diversity penalty on repeated aspects.
      \end{itemize}
      \[
        IG_{\mathrm{final}}(q)=IG(q)-\lambda\max_{q'\in Q_{\mathrm{hist}}}\!\mathrm{Sim}(q,q').
      \]
      \textbf{Retained as design}; not evaluated as implemented modules.
    \end{column}
  \end{columns}
  \source{Supplied SEMSYNTH manuscript; course report, Method}
\end{frame}

\begin{frame}{4. Information gain in the implemented special case}
  \[
    IG(q)=H(\mathcal H)-\sum_{a\in\{\mathrm{Yes,No}\}}P(a\mid q)H(\mathcal H\mid a,q).
  \]
  \textbf{Assumptions:} uniform priors, deterministic probe execution, exact feedback.
  \par\medskip
  For the question ``On input $x$, should the output equal $y$?'', let
  $n=|\mathcal H|$ and $p=|\mathcal H_{\mathrm{Yes}}|/n$. For nonempty parts,
  \begin{align*}
    IG(q)&=\log_2 n-p\log_2(pn)-(1-p)\log_2((1-p)n)\\
         &=-p\log_2p-(1-p)\log_2(1-p)=H_b(p).
  \end{align*}
  \begin{block}{Meaning for the prototype}
    Balanced selection maximizes this score over available probe partitions.
    With three candidates, a 1--2 split is the closest available balance.
    Bounded probes can still leave candidates unresolved.
  \end{block}
  \source{Course report, Method: derivation and conditional query bound}
\end{frame}

\begin{frame}{7. Controlled evaluation protocol}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Shared ordered pool per task}
      \begin{itemize}
        \item Direct-1: first candidate, no questions.
        \item First-valid: first admitted candidate, no questions.
        \item Balanced: entropy-based questions and hard filtering.
      \end{itemize}
      \par\medskip
      Seed 3101, requested $K=3$, at most three questions and eight probe inputs. Saved pilot pools contain two or three candidates.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Exact-input separation}
      \[
        X_{\mathrm{eval,clean}}=X_{\mathrm{eval}}\setminus X_{\mathrm{probes}}.
      \]
      \begin{itemize}
        \item Offline: 33 of 160 input instances excluded; 127 retained.
        \item Each eight-case pilot: 31 retained input instances.
        \item Success means passing all retained tests for a case.
      \end{itemize}
      \par\medskip
      Reference feedback is noiseless. Input separation does not establish generalization to new task families.
    \end{column}
  \end{columns}
  \source{Saved run manifests/checkpoints; course report, Experimental Setup}
\end{frame}

\begin{frame}{11. Conclusions and remaining original modules}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Implemented and measured}
      \begin{itemize}
        \item Execution-based binary clarification and hard filtering on small pools.
        \item 32/32 controlled offline successes with a reference in every bank.
        \item Saved pilot selection: 8/8 per identifier on retained finite test suites.
      \end{itemize}
      \par\medskip
      Results support this subset under the stated protocol. Human effort, noisy feedback, and broad generation accuracy remain unmeasured.
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Original design retained for future integration}
      \begin{itemize}
        \item Train and calibrate 47-feature $M(h,q)$.
        \item Integrate atomic questions and diversity.
        \item Add consistency recovery, two-stage AGA, and regeneration.
        \item Evaluate larger spaces and real users.
      \end{itemize}
